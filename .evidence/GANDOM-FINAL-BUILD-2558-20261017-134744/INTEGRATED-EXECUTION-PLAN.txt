const crypto = require('crypto');
const WebSocket = require('ws');

const NODE_ID = process.env.GANDOM_NODE_ID;
const DISCOVERY_URL = process.env.GANDOM_DISCOVERY_URL || '';

if (!NODE_ID) {
  throw new Error('GANDOM_NODE_ID_REQUIRED');
}

const identity = crypto
  .createHash('sha256')
  .update(`GANDOM-DISCOVERY:${NODE_ID}`)
  .digest('hex');

function discover() {
  if (!DISCOVERY_URL) {
    return {
      ok: false,
      reason: 'DISCOVERY_ENDPOINT_NOT_CONFIGURED',
      node_id: NODE_ID,
      identity
    };
  }

  return new Promise((resolve) => {
    const ws = new WebSocket(DISCOVERY_URL, {
      headers: {
        'x-gandom-node': NODE_ID
      }
    });

    const timer = setTimeout(() => {
      try { ws.close(); } catch (_) {}
      resolve({
        ok: false,
        reason: 'DISCOVERY_TIMEOUT',
        node_id: NODE_ID,
        identity
      });
    }, 10000);

    ws.once('open', () => {
      ws.send(JSON.stringify({
        type: 'DISCOVERY_REQUEST',
        node_id: NODE_ID,
        identity,
        protocol: 'GANDOM-P2P-DISCOVERY-1'
      }));
    });

    ws.once('message', (raw) => {
      clearTimeout(timer);

      let message;
      try {
        message = JSON.parse(raw.toString());
      } catch (_) {
        resolve({
          ok: false,
          reason: 'INVALID_DISCOVERY_RESPONSE',
          node_id: NODE_ID,
          identity
        });
        try { ws.close(); } catch (_) {}
        return;
      }

      resolve({
        ok: true,
        node_id: NODE_ID,
        identity,
        response: message
      });

      try { ws.close(); } catch (_) {}
    });

    ws.once('error', (err) => {
      clearTimeout(timer);
      resolve({
        ok: false,
        reason: 'DISCOVERY_CONNECTION_ERROR',
        error: err.message,
        node_id: NODE_ID,
        identity
      });
    });
  });
}

module.exports = {
  discover,
  NODE_ID,
  identity

  const peerId = `outbound:${PEER_ADDRESS}`;

  ws.on('open', () => {
    if (shuttingDown) {
      try { ws.close(); } catch {}
      return;
    }

    peers.set(peerId, {
      ws,
      direction: 'outbound',
      address: PEER_ADDRESS,
      connectedAt: Date.now()
    });

    log('PEER_OUTBOUND_CONNECTED', {
      peer_id: peerId,
      address: PEER_ADDRESS,
      peers: peers.size
    });

    send(ws, 'HELLO', {
      node_id: NODE_ID,
      direction: 'outbound'
    });
  });

  ws.on('message', raw => {
    let message;

    try {
      message = JSON.parse(raw.toString());
    } catch (error) {
      log('PEER_OUTBOUND_INVALID_MESSAGE', {
        peer_id: peerId,
        error: error.message
      });
      return;
    }

    log('PEER_OUTBOUND_MESSAGE', {
      peer_id: peerId,
      type: message.type || null
    });

    if (message.type === 'PING') {
      send(ws, 'PONG', {
        node_id: NODE_ID,
        uptime: process.uptime()
      });
      return;
    }

    if (message.type === 'HELLO_ACK') {
      log('PEER_OUTBOUND_HELLO_ACK', {
        peer_id: peerId,
        remote_node_id:
          message.payload && message.payload.node_id
            ? message.payload.node_id
            : null
      });
      return;
    }

    if (message.type === 'DATA_ACK') {
      log('PEER_OUTBOUND_DATA_ACK', {
        peer_id: peerId,
        accepted:
          message.payload && typeof message.payload.accepted !== 'undefined'
            ? message.payload.accepted
            : null
      });
      return;
    }

    if (message.type === 'PONG') {
      log('PEER_OUTBOUND_PONG', {
        peer_id: peerId
      });
    }
  });

  ws.on('close', () => {
    peers.delete(peerId);

    if (outboundPeer === ws) {
      outboundPeer = null;
    }

    log('PEER_OUTBOUND_DISCONNECTED', {
      peer_id: peerId,
      peers: peers.size,
      reconnect_in_ms: shuttingDown ? 0 : RECONNECT_DELAY
    });

    if (!shuttingDown && !reconnectTimer) {
      reconnectTimer = setTimeout(() => {
        reconnectTimer = null;
        connectOutbound();
      }, RECONNECT_DELAY);
    }
  });

  ws.on('error', error => {
    log('PEER_OUTBOUND_ERROR', {
      peer_id: peerId,
      address: PEER_ADDRESS,
      error: error.message
    });
  });
}
