function run(file) {
  const full = path.join(root, file);
  if (!fs.existsSync(full)) return null;

  const r = spawnSync(process.execPath, [full], {
    cwd: path.join(root, '..', '..'),
    encoding: 'utf8',
    timeout: 300000
  });

  return {
    exitCode: r.status,
    output: `${r.stdout || ''}${r.stderr || ''}`
  };
}
